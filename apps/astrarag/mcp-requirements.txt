# Deps for the MCP venv (/opt/mcp/venv). A file, not a Dockerfile ARG: the image
# has no shell, so RUN is exec form, which does not expand variables.

# Prerelease of the 4.x line (3.4.5 is the newest stable) on purpose: the first
# line whose SSRF guard can be told to trust our egress proxy, which is what lets
# CIMD clients attach at all. Gated on FASTMCP_SSRF_TRUST_PROXY, set on the
# service. Pulls MCP SDK 2.x and httpx2; our own `import httpx` keeps resolving to
# the 0.28 line via llama-cloud. Needs starlette >=1.0.1 — the reason this venv
# exists at all.
fastmcp==4.0.0b1

# Current supported line, not a bump of the 0.1.41 upstream used to ship: different
# generator, different method names. Covers Index V1+V2, retrieval, parse/classify/
# extract and job polling. Still raw HTTP for split and pipeline files — this SDK
# targets the deprecated beta routes there.
llama-cloud==2.13.0

# The OAuth broker's state — client registrations, authorize states, codes and
# refresh tokens — lives in the platform's own Redis, so it survives a restart
# and a second replica. Nothing else in the venv brings a Redis driver.
redis==8.0.1
